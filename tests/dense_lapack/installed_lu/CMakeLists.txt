cmake_minimum_required(VERSION 3.25)
project(ASCCppInstalledLuFamilies LANGUAGES CXX)
string(REPLACE "|" ";" ASC_CPP_LAPACK_RUNTIME_LIBRARIES "${TEST_RUNTIMES}")
find_package(ASCCpp 0.9 CONFIG REQUIRED COMPONENTS dense_lapack)
get_property(_languages GLOBAL PROPERTY ENABLED_LANGUAGES)
if(NOT _languages STREQUAL "CXX")
  message(FATAL_ERROR "Installed provider enabled foreign languages")
endif()
foreach(_forbidden IN ITEMS sparse random utilities core_cuda dense_cuda)
  if(TARGET "ASC::${_forbidden}")
    message(FATAL_ERROR "Dense LAPACK lookup leaked ASC::${_forbidden}")
  endif()
endforeach()
if(NOT ASCCpp_LAPACK_PROFILE STREQUAL "incremental-lu-v2")
  message(FATAL_ERROR "Unexpected installed development profile")
endif()
add_executable(installed_lu main.cc)
target_link_libraries(installed_lu PRIVATE ASC::dense_lapack)
enable_testing()
add_test(NAME installed_lu_families COMMAND installed_lu)
