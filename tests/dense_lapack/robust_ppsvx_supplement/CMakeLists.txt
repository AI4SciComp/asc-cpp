cmake_minimum_required(VERSION 3.25)
project(ASCCppRobustPpsvxSupplement LANGUAGES CXX)

# Supplemental compilation of the actual first-party adapter against an
# installed admitted provider context. This is not provider-platform admission.
find_package(ASCCpp 0.9 CONFIG REQUIRED COMPONENTS dense_lapack)
find_package(Threads REQUIRED)
if(NOT ASCCpp_EXPERIMENTAL_ROBUST_PPSVX)
  message(FATAL_ERROR "An experimental robust PPSVX installation is required")
endif()
option(ASC_ROBUST_OBSERVE_LOADS "Calibrated Clang load observation" OFF)
option(ASC_ROBUST_SANITIZERS "ASan/UBSan on adapter and test instructions" OFF)
if(NOT CMAKE_CXX_COMPILER_ID MATCHES "^(GNU|Clang)$")
  message(FATAL_ERROR "This supplemental harness requires GNU or Clang")
endif()
get_filename_component(_source "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
file(READ "${ASCCpp_DIR}/ASCCppLapackProvider.json" _metadata)
string(JSON _bits GET "${_metadata}" integer_bits)
string(JSON _source_digest GET "${_metadata}" source_input_sha256)
string(JSON _build_digest GET "${_metadata}" identity_sha256)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/lapack_build_config.h"
  "#define ASC_LAPACK_INTEGER_BITS ${_bits}\n"
  "#define ASC_LAPACK_SOURCE_SHA256 \"${_source_digest}\"\n"
  "#define ASC_LAPACK_BUILD_SHA256 \"${_build_digest}\"\n")

add_library(adapter STATIC "${_source}/src/dense/lapack/robust_cholesky_packed_expert.cc")
target_compile_features(adapter PUBLIC cxx_std_20)
target_include_directories(adapter PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_include_directories(adapter SYSTEM PRIVATE "${ASC_CPP_LAPACK_ROOT}/include")
target_link_libraries(adapter PUBLIC ASC::dense_lapack Threads::Threads)
target_compile_options(adapter PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(ASC_ROBUST_SANITIZERS)
  target_compile_options(adapter PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(adapter PUBLIC -fsanitize=address,undefined)
endif()
add_executable(math "${_source}/tests/dense_lapack/robust_packed_cholesky_expert_math_test.cc")
target_include_directories(math PRIVATE "${_source}/tests/dense_lapack/installed_lu")
target_link_libraries(math PRIVATE adapter)
add_executable(consumer "${_source}/tests/dense_lapack/installed_lu/robust_packed_cholesky_expert_main.cc")
target_link_libraries(consumer PRIVATE adapter)
enable_testing()
foreach(_scalar IN ITEMS s d c z)
  add_test(NAME "robust_ppsvx.supplement.required_math_${_scalar}" COMMAND math "${_scalar}")
  add_test(NAME "robust_ppsvx.supplement.general_${_scalar}" COMMAND consumer "--general-${_scalar}")
endforeach()
foreach(_mode IN ITEMS ordinary concurrent safety)
  add_test(NAME "robust_ppsvx.supplement.${_mode}" COMMAND consumer "--${_mode}")
endforeach()
if(ASC_ROBUST_OBSERVE_LOADS)
  if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
    message(FATAL_ERROR "trace-loads observation requires Clang")
  endif()
  # The observer has no load-coverage instrumentation. Only the actual adapter
  # and consumer instructions are observed; installed/foreign code is not.
  add_library(observer OBJECT "${_source}/tests/array_io/printing_load/observation.cc")
  target_compile_features(observer PRIVATE cxx_std_20)
  target_compile_options(adapter PRIVATE -O1 -fsanitize-coverage=trace-pc,trace-loads)
  target_compile_options(consumer PRIVATE -O1 -fsanitize-coverage=trace-pc,trace-loads)
  target_compile_definitions(consumer PRIVATE ASC_ROBUST_PPSVX_LOAD_OBSERVATION)
  target_include_directories(consumer PRIVATE "${_source}/tests/array_io/printing_load")
  # The adapter's load callbacks are also needed by the mathematical executable.
  target_sources(consumer PRIVATE $<TARGET_OBJECTS:observer>)
  target_sources(math PRIVATE $<TARGET_OBJECTS:observer>)
endif()
get_property(_tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${_tests} PROPERTIES TIMEOUT 120)
