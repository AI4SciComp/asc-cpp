cmake_minimum_required(VERSION 3.25)
project(ASCCppInstalledFileCloseTests LANGUAGES CXX)
include(CTest)

# This adapter exercises the actual installed Core File implementation using
# GNU ELF wrapping. Unsupported linkage/platforms are errors, never skips.
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR
   NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  message(FATAL_ERROR "The retained close-fault adapter requires Linux GNU ELF wrapping.")
endif()
if(NOT DEFINED ASC_FILE_CLOSE_FIXTURES)
  message(FATAL_ERROR "ASC_FILE_CLOSE_FIXTURES must name prepared external fixtures.")
endif()
if(NOT DEFINED ASC_FILE_CLOSE_COMPONENT OR
   NOT ASC_FILE_CLOSE_COMPONENT MATCHES "^(dense|sparse)$")
  message(FATAL_ERROR "ASC_FILE_CLOSE_COMPONENT must select dense or sparse.")
endif()
find_package(ASCCpp 0.9.0 EXACT CONFIG REQUIRED COMPONENTS ${ASC_FILE_CLOSE_COMPONENT})
get_target_property(_core_type ASC::core TYPE)
if(NOT _core_type STREQUAL "STATIC_LIBRARY")
  message(FATAL_ERROR "GNU --wrap does not intercept calls inside a shared Core library.")
endif()
if(ASC_FILE_CLOSE_COMPONENT STREQUAL "dense")
  set(_unrelated sparse)
else()
  set(_unrelated dense)
endif()
foreach(_target IN ITEMS ${_unrelated} dense_lapack random utilities core_cuda)
  if(TARGET "ASC::${_target}")
    message(FATAL_ERROR "Isolated close test imported ${_target}.")
  endif()
endforeach()
include("${ASC_FILE_CLOSE_FIXTURES}/cases.cmake")
list(LENGTH ASC_FILE_CLOSE_CASES _case_count)
if(NOT _case_count EQUAL 60)
  message(FATAL_ERROR "Expected the complete 60-case cleanup matrix.")
endif()

function(_close_test name source)
  add_executable("${name}" "${source}")
  target_compile_features("${name}" PRIVATE cxx_std_20)
  set_target_properties("${name}" PROPERTIES CXX_EXTENSIONS OFF)
  target_compile_options("${name}" PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_link_libraries("${name}" PRIVATE "ASC::${ASC_FILE_CLOSE_COMPONENT}")
  target_link_options("${name}" PRIVATE LINKER:--wrap=fclose LINKER:--wrap=fflush)
  if(ASC_FILE_CLOSE_COMPONENT STREQUAL "dense")
    target_compile_definitions("${name}" PRIVATE TEST_DENSE=1)
  endif()
  set(_scratch "${CMAKE_CURRENT_BINARY_DIR}/scratch/${name}")
  file(MAKE_DIRECTORY "${_scratch}")
  add_test(NAME "asc_cpp.array_io.${name}" COMMAND "${name}")
  set_tests_properties("asc_cpp.array_io.${name}" PROPERTIES
    WORKING_DIRECTORY "${_scratch}" LABELS "array-io;file-cleanup;installed")
endfunction()

foreach(_case IN LISTS ASC_FILE_CLOSE_CASES)
  if((ASC_FILE_CLOSE_COMPONENT STREQUAL "dense" AND _case MATCHES "dense") OR
     (ASC_FILE_CLOSE_COMPONENT STREQUAL "sparse" AND NOT _case MATCHES "dense"))
    set(_name "file_close_${_case}")
    _close_test("${_name}" main.cc)
    target_include_directories("${_name}" PRIVATE "${ASC_FILE_CLOSE_FIXTURES}/${_case}")
  endif()
endforeach()
_close_test("file_close_matrix_market_${ASC_FILE_CLOSE_COMPONENT}" matrix_market.cc)
